};

  const [state, setState] = useState(INITIAL_STATE);
  const [hovered, setHovered] = useState(false);

      setHovered(false);
    }, 300);

  return (

      {/* Background Sparkle Trails */}
      <div className="absolute top-1/4 left-1/4 right-1/4 bottom-1/4 flex flex-col justify-between items-center pointer-events-none">
        {Array.from({ length: 12 }).map((_, i) => (
          <div key={i} className="w-10 h-10 bg-white/10 animate-pulse rounded-full shadow-lg shadow-indigo-500/30" />
        ))}
      </div>

      {/* Top Border Gradient */}
      <div className="absolute top-0 left-0 right-0 h-1 bg-clip-padding from-indigo-600 via-purple-600 to-pink-600" />

      {/* Inner Glassmorphism Container */}
      <div className="relative max-w-7xl mx-auto p-8 sm:p-12">

                </div>

          </div>

      </div>

      {/* Bottom Gradient Border */}
      <div className="absolute bottom-0 left-0 right-0 h-1 bg-clip-padding from-white/10 via-white/5 to-white/0" />

      {/* Mouse Hover Effects */}
      <div className="absolute -top-1/4 -right-1/4 -bottom-1/4 -left-1/4 bg-white/5 rounded-xl blur opacity-0 transition-all duration-500 group-hover:opacity-20" />
